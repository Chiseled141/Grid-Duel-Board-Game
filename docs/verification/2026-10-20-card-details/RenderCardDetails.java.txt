package deckduel.client.ui;
import deckduel.cards.CatalogLoader;
import deckduel.client.ClientSettings;
import deckduel.client.ScoringSounds;
import deckduel.core.card.*;
import deckduel.net.MatchUpdate;
import java.awt.Rectangle;
import java.io.File;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import javafx.application.Platform;
import javafx.embed.swing.SwingFXUtils;
import javafx.geometry.Point2D;
import javafx.scene.Scene;
import javax.imageio.ImageIO;
public class RenderCardDetails {
 static final Catalog C=CatalogLoader.loadBundled(); static final List<CardDef> D=DeckRules.deck(C);
 static MatchState chosen(long seed) {
  var s=MatchState.start(D,D,C.landmarks(),seed);
  for(Seat seat:Seat.values())s=s.apply(seat,GreedyBot.choose(s.view(seat)));
  return s;
 }
 static MatchState watch() {
  for(long seed=1;seed<100;seed++) {
   var s=MatchState.start(D,D,C.landmarks(),seed);
   var offers=s.view(Seat.FIRST).offers();
   var w=offers.stream().filter(c->c.def().id().equals("watchtower")).findFirst();
   if(w.isEmpty())continue;
   s=s.apply(Seat.FIRST,new PickLandmarks(List.of(w.get().id(),offers.stream().filter(c->c.id()!=w.get().id()).findFirst().orElseThrow().id())));
   return s.apply(Seat.SECOND,GreedyBot.choose(s.view(Seat.SECOND)));
  }
  throw new AssertionError();
 }
 static MatchState spy() {
  for(long seed=1;seed<100;seed++) {
   var s=chosen(seed);var c=s.view(Seat.FIRST).hand().stream().filter(x->x.def().effect() instanceof Effect.Spy).findFirst();
   if(c.isEmpty())continue;
   if(s.turn()==Seat.SECOND)s=s.apply(Seat.SECOND,new SkipTurn());
   return s.apply(Seat.FIRST,PlayCard.cast(c.get().id(),null));
  }
  throw new AssertionError();
 }
 static void click(CardTableView t,Rectangle r){var p=new Point2D(r.getCenterX(),r.getCenterY());t.pressAt(p);t.releaseAt(p);}
 static void show(CardTableView t,PlayerView v){t.show(new CardTableView.State(new MatchUpdate("Preview",1,"You","Practice bot",true,true,false,false,false,v),false,true));}
 static void shot(CardTableView t,String path)throws Exception{t.applyCss();t.layout();ImageIO.write(SwingFXUtils.fromFXImage(t.snapshot(null,null),null),"png",new File(path));}
 public static void main(String[] args)throws Exception{
  FxKit.start();CardImages.shared().preload(C.cards().stream().map(CardDef::id).toList());Thread.sleep(1000);
  var done=new CompletableFuture<Void>();
  Platform.runLater(()->{try{
   for(int[] size:new int[][]{{1280,720},{1470,920}}) {
    var t=new CardTableView(new CardTableView.Actions(){public void move(Move m){}public void rematch(){}public void leave(){}},new ClientSettings(),new ScoringSounds(()->false));
    new Scene(t,size[0],size[1]).getStylesheets().add(FxKit.STYLESHEET);
    ((javafx.scene.image.ImageView)t.getChildren().get(0)).setImage(new javafx.scene.image.Image(RenderCardDetails.class.getResource("/assets/match-room.png").toExternalForm(),false));
    t.resize(size[0],size[1]);t.applyCss();t.layout();var v=watch().view(Seat.FIRST);show(t,v);t.shown(0);
    shot(t,args[0]+"/dock-"+size[0]+".png");
    int index=0;while(!v.you().landmarks().get(index).def().id().equals("watchtower"))index++;
    click(t,t.landmarkBounds(TableZone.OWN_LANDMARK,index));shot(t,args[0]+"/buff-"+size[0]+".png");
    t.show(new CardTableView.State(null,false,true));show(t,spy().view(Seat.FIRST));
    click(t,t.revealedBounds(0));shot(t,args[0]+"/spy-"+size[0]+".png");t.hidden();
   }
   done.complete(null);
  }catch(Throwable e){done.completeExceptionally(e);}});
  try{done.get();}finally{Platform.exit();}System.exit(0);
 }
}
