package deckduel.client.ui;
import deckduel.cards.CatalogLoader;
import deckduel.client.ClientSettings;
import deckduel.client.ScoringSounds;
import deckduel.core.card.*;
import deckduel.net.MatchUpdate;
import java.io.File;
import java.util.concurrent.CompletableFuture;
import javafx.application.Platform;
import javafx.embed.swing.SwingFXUtils;
import javafx.scene.Scene;
import javax.imageio.ImageIO;
public class RenderSkipHand {
 public static void main(String[] args) throws Exception {
  var catalog=CatalogLoader.loadBundled(); var deck=DeckRules.deck(catalog);
  MatchState state=MatchState.start(deck,deck,catalog.landmarks(),3);
  for(Seat seat:Seat.values()) state=state.apply(seat,GreedyBot.choose(state.view(seat)));
  for(int i=0;i<6;i++) state=state.apply(state.turn(),state.view(state.turn()).legalPlays().get(0));
  PlayerView v=state.view(state.turn());
  FxKit.start();
  CardImages.shared().preload(catalog.cards().stream().map(CardDef::id).toList());
  Thread.sleep(1000);
  var done=new CompletableFuture<Void>();
  Platform.runLater(()->{try {
   for(int[] size:new int[][]{{1280,720},{1470,920}}) {
    var view=new CardTableView(new CardTableView.Actions(){public void move(Move m){}public void rematch(){}public void leave(){}},new ClientSettings(),new ScoringSounds(()->false));
    new Scene(view,size[0],size[1]).getStylesheets().add(FxKit.STYLESHEET);
    ((javafx.scene.image.ImageView)view.getChildren().get(0)).setImage(new javafx.scene.image.Image(RenderSkipHand.class.getResource("/assets/match-room.png").toExternalForm(), false));
    view.resize(size[0],size[1]); view.applyCss(); view.layout();
    view.show(new CardTableView.State(new MatchUpdate("Preview",1,"You","Practice bot",true,true,false,false,false,v),false,true));
    view.shown(0);
    ImageIO.write(SwingFXUtils.fromFXImage(view.snapshot(null,null),null),"png",new File(args[0]+"/table-"+size[0]+".png"));
    var row=view.slotBounds(TableZone.OWN,0); var hand=view.handSlot(0,v.hand().size());
    System.out.println(size[0]+"x"+size[1]+": row "+row.width+"x"+row.height+", hand "+hand.width+"x"+hand.height);
    view.hidden();
   }
   done.complete(null);
  }catch(Throwable e){done.completeExceptionally(e);}});
  try {done.get();} finally {Platform.exit();}
  System.exit(0);
 }
}
