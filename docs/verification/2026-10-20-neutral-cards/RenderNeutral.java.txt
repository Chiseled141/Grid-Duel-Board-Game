package deckduel.client.ui;

import deckduel.cards.CatalogLoader;
import deckduel.client.ClientSettings;
import deckduel.client.ScoringSounds;
import deckduel.core.card.*;
import deckduel.net.MatchUpdate;
import java.io.File;
import java.util.concurrent.CompletableFuture;
import javafx.application.Platform;
import javafx.embed.swing.SwingFXUtils;
import javafx.scene.Scene;
import javafx.stage.Stage;
import javax.imageio.ImageIO;

/** Throwaway: renders the new-rules table states to PNGs for visual review. */
public class RenderNeutral {
    static Stage stage;
    static CardTableView view;

    interface MoveTest { boolean test(Move move, MatchState before, MatchState after); }

    static MatchState[] find(MoveTest wanted) {
        var catalog = CatalogLoader.loadBundled();
        var deck = DeckRules.deck(catalog);
        for (long seed = 1; seed < 400; seed++) {
            MatchState state = MatchState.start(deck, deck, catalog.landmarks(), seed);
            while (state.phase() != MatchPhase.FINISHED) {
                MatchState before = state;
                Seat seat = state.awaits(Seat.FIRST) ? Seat.FIRST : Seat.SECOND;
                Move move = GreedyBot.choose(state.view(seat));
                state = state.apply(seat, move);
                if (wanted.test(move, before, state)) return new MatchState[]{before, state};
            }
        }
        throw new AssertionError();
    }

    static void fx(Runnable r) throws Exception {
        CompletableFuture<Void> done = new CompletableFuture<>();
        Platform.runLater(() -> { try { r.run(); done.complete(null); } catch (Throwable t) { done.completeExceptionally(t); } });
        done.get();
    }

    static void shot(String name) throws Exception {
        fx(() -> {
            try { ImageIO.write(SwingFXUtils.fromFXImage(stage.getScene().getRoot().snapshot(null, null), null), "png", new File(name)); }
            catch (Exception e) { throw new RuntimeException(e); }
        });
    }

    static CardTableView.State st(PlayerView v, boolean reduced) {
        return new CardTableView.State(new MatchUpdate("M1", 1, "alice", "bob", false, true, false, false, false, v), false, reduced);
    }

    static void still(String name, PlayerView v) throws Exception {
        fx(() -> view.show(new CardTableView.State(null, false, true)));
        fx(() -> view.show(st(v, true)));
        Thread.sleep(400);
        shot(name);
    }

    public static void main(String[] args) throws Exception {
        String out = args[0];
        FxKit.start();
        CardImages.shared().preload(CatalogLoader.loadBundled().cards().stream().map(CardDef::id).toList());
        Thread.sleep(1500);
        fx(() -> {
            view = new CardTableView(new CardTableView.Actions() { public void move(Move m) { } public void rematch() { } public void leave() { } }, new ClientSettings(), new ScoringSounds(() -> false));
            stage = new Stage();
            Scene scene = new Scene(view, 1366, 800);
            scene.getStylesheets().add(FxKit.STYLESHEET);
            stage.setScene(scene);
            stage.show();
            view.shown(0);
        });
        var catalog = CatalogLoader.loadBundled();
        var deck = DeckRules.deck(catalog);
        MatchState choosing = MatchState.start(deck, deck, catalog.landmarks(), 4);
        still(out + "/1-choose.png", choosing.view(Seat.FIRST));
        MatchState[] mid = find((m, b, a) -> a.phase() == MatchPhase.IN_PROGRESS && a.round() == 1
                && a.view(Seat.FIRST).you().row().cards().size() >= 4 && a.view(Seat.FIRST).yourTurn());
        still(out + "/2-table.png", mid[1].view(Seat.FIRST));
        MatchState[] recon = find((m, b, a) -> !a.view(Seat.FIRST).looking().isEmpty());
        still(out + "/3-recon.png", recon[1].view(Seat.FIRST));
        MatchState[] spy = find((m, b, a) -> !a.view(Seat.FIRST).revealed().isEmpty() && a.view(Seat.FIRST).yourTurn());
        still(out + "/4-spy.png", spy[1].view(Seat.FIRST));
        MatchState[] sab = find((m, b, a) -> m instanceof PlayCard && a.view(Seat.FIRST).lastPlay() != null
                && a.view(Seat.FIRST).lastPlay().card().def().effect() instanceof Effect.Sabotage
                && a.view(Seat.FIRST).lastPlay().actor() == Seat.FIRST && a.round() == b.round());
        fx(() -> view.show(new CardTableView.State(null, false, true)));
        fx(() -> view.show(st(sab[0].view(Seat.FIRST), false)));
        Thread.sleep(4500);
        fx(() -> view.show(st(sab[1].view(Seat.FIRST), false)));
        long start = System.nanoTime();
        for (int t : new int[]{900, 1250, 1500, 1900}) {
            Thread.sleep(Math.max(0, Math.round(t * Staging.TEMPO) - (System.nanoTime() - start) / 1_000_000));
            shot(out + "/5-sabotage-" + t + ".png");
        }
        System.exit(0);
    }
}
